```python
from __future__ import annotations

import hashlib
import json
import os
import tempfile
from pathlib import Path
from typing import Any

from .errors import ValidationError


def ensure_inside(root: Path, candidate: Path) -> Path:
    root = root.resolve()
    candidate = candidate.resolve()
    try:
        candidate.relative_to(root)
    except ValueError as exc:
        raise ValidationError(f"path escapes project root: {candidate}") from exc
    return candidate


def read_json(path: Path) -> dict[str, Any]:
    try:
        value = json.loads(path.read_text(encoding="utf-8"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise ValidationError(f"invalid JSON file: {path}") from exc
    if not isinstance(value, dict):
        raise ValidationError(f"JSON root must be an object: {path}")
    return value


def atomic_write_json(path: Path, payload: dict[str, Any]) -> None:
    data = json.dumps(payload, ensure_ascii=False, indent=2, sort_keys=True) + "\n"
    atomic_write_text(path, data)


def atomic_write_text(path: Path, data: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    data = data.replace("\r\n", "\n").replace("\r", "\n")
    temporary_name: str | None = None
    try:
        with tempfile.NamedTemporaryFile(
            "w",
            encoding="utf-8",
            newline="\n",
            dir=path.parent,
            prefix=f".{path.name}.",
            suffix=".tmp",
            delete=False,
        ) as handle:
            temporary_name = handle.name
            handle.write(data)
            handle.flush()
            os.fsync(handle.fileno())
        os.replace(temporary_name, path)
    finally:
        if temporary_name and os.path.exists(temporary_name):
            os.unlink(temporary_name)


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for block in iter(lambda: handle.read(65536), b""):
            digest.update(block)
    return digest.hexdigest()
```

Create `skills/orchestra/scripts/orchestra_lib/model.py`:

```python
from __future__ import annotations

import copy
import re
from pathlib import PurePosixPath
from typing import Any

from .constants import (
    CONFIRMATION_POLICIES,
    CORE_AGENTS,
    LEGAL_TRANSITIONS,
    LEVELS,
    SCHEMA_VERSION,
)
from .errors import ConflictError, ValidationError

TASK_ID = re.compile(r"^TASK-\d{8}-\d{3}$")
TASK_KEYS = {
    "schema_version", "task_id", "revision", "title", "level", "goal", "approved_basis",
    "allowed_changes", "forbidden_changes", "relevant_files",
    "acceptance_criteria", "required_checks", "risk_tags", "agents",
    "agent_overrides", "confirmation_policy", "git",
}
CONFIG_KEYS = {
    "schema_version", "confirmation_policy", "retain_task_history",
    "enabled_specialists", "max_relevant_files", "protected_paths",
    "command_source_files",
}


def _require_exact_keys(value: dict[str, Any], expected: set[str], label: str) -> None:
    missing = sorted(expected - value.keys())
    extra = sorted(value.keys() - expected)
    if missing or extra:
        raise ValidationError(f"{label} keys mismatch: missing={missing}, extra={extra}")


def _string_list(value: Any, label: str) -> list[str]:
    if not isinstance(value, list) or not all(isinstance(item, str) and item for item in value):
        raise ValidationError(f"{label} must be a list of non-empty strings")
    return value


def _repo_path(value: str, label: str) -> None:
    if "\\" in value or ":" in value or value.startswith("/"):
        raise ValidationError(f"{label} must use repository-relative POSIX form: {value}")
    if "*" in value.removesuffix("/**"):
        raise ValidationError(f"{label} supports only a trailing /** subtree marker: {value}")
    parts = PurePosixPath(value.removesuffix("/**")).parts
    if not parts or ".." in parts:
        raise ValidationError(f"{label} escapes repository scope: {value}")


def validate_config(config: dict[str, Any]) -> None:
    _require_exact_keys(config, CONFIG_KEYS, "config")
    if config["schema_version"] != SCHEMA_VERSION:
        raise ValidationError("unsupported config schema_version")
    if config["confirmation_policy"] not in CONFIRMATION_POLICIES:
        raise ValidationError("invalid confirmation_policy")
    if not isinstance(config["retain_task_history"], bool):
        raise ValidationError("retain_task_history must be boolean")
    _string_list(config["enabled_specialists"], "enabled_specialists")
    _string_list(config["protected_paths"], "protected_paths")
    _string_list(config["command_source_files"], "command_source_files")
    if not isinstance(config["max_relevant_files"], int) or config["max_relevant_files"] < 1:
        raise ValidationError("max_relevant_files must be a positive integer")


def validate_task(task: dict[str, Any], config: dict[str, Any]) -> None:
    validate_config(config)
    _require_exact_keys(task, TASK_KEYS, "task")
    if task["schema_version"] != SCHEMA_VERSION:
        raise ValidationError("unsupported task schema_version")
    if not TASK_ID.fullmatch(task["task_id"]):
        raise ValidationError("invalid task_id")
    if not isinstance(task["revision"], int) or task["revision"] < 1:
        raise ValidationError("revision must be a positive integer")
    if task["level"] not in LEVELS:
        raise ValidationError("invalid task level")
    for key in ("title", "goal"):
        if not isinstance(task[key], str) or not task[key].strip():
            raise ValidationError(f"{key} must be non-empty")
    if set(task["approved_basis"]) != {"product", "technical_plan"}:
        raise ValidationError("approved_basis requires product and technical_plan")
    for path in task["approved_basis"].values():
        if path:
            _repo_path(path, "approved_basis")
    for key in (
        "allowed_changes", "forbidden_changes", "relevant_files",
        "acceptance_criteria", "required_checks", "risk_tags", "agents",
    ):
        _string_list(task[key], key)
    for key in ("allowed_changes", "forbidden_changes", "relevant_files"):
        for path in task[key]:
            _repo_path(path, key)
    if len(task["relevant_files"]) > config["max_relevant_files"] and task["level"] != "MAJOR":
        raise ValidationError("relevant_files exceeds configured limit")
    overrides = task["agent_overrides"]
    if set(overrides) != {"add", "remove", "reason"}:
        raise ValidationError("agent_overrides keys mismatch")
    _string_list(overrides["add"], "agent_overrides.add")
    _string_list(overrides["remove"], "agent_overrides.remove")
    if not isinstance(overrides["reason"], str):
        raise ValidationError("agent_overrides.reason must be a string")
    removed_core = set(CORE_AGENTS[task["level"]]) & set(overrides["remove"])
    if removed_core and not overrides["reason"].strip():
        raise ValidationError("removing a core Agent requires a reason")
    if task["confirmation_policy"] not in CONFIRMATION_POLICIES:
        raise ValidationError("invalid task confirmation_policy")
    git = task["git"]
    if set(git) != {"base_branch", "task_branch", "worktree", "code_base_commit"}:
        raise ValidationError("git keys mismatch")
    if not all(isinstance(git[key], str) and git[key] for key in git):
        raise ValidationError("git values must be non-empty strings")


def new_state(task: dict[str, Any]) -> dict[str, Any]:
    return {
        "schema_version": SCHEMA_VERSION,
        "task_id": task["task_id"],
        "stage": "DRAFT",
        "active_agent": None,
        "next_event": "request_task_approval",
        "iterations": {"coding": 0, "review": 0},
        "approvals": {},
        "evidence": {},
        "generated_files": {},
        "failures": {},
        "blocker": None,
    }


def apply_transition(state: dict[str, Any], event: str) -> dict[str, Any]:
    target = LEGAL_TRANSITIONS.get((state["stage"], event))
    if target is None:
        raise ConflictError(f"illegal transition: {state['stage']} + {event}")
    changed = copy.deepcopy(state)
    changed["stage"] = target
    changed["next_event"] = None
    changed["blocker"] = None
    return changed
```

```python
from __future__ import annotations

from pathlib import Path
from typing import Any

from .errors import ConflictError
from .io import atomic_write_json, atomic_write_text, ensure_inside, read_json
from .model import new_state, validate_config, validate_task


class OrchestraService:
    def __init__(self, project_root: Path, assets_root: Path):
        self.project_root = project_root.resolve()
        self.assets_root = assets_root.resolve()
        self.control_root = self.project_root / ".orchestra"

    @property
    def config_path(self) -> Path:
        return self.control_root / "config.json"

    def _default_config(self) -> dict[str, Any]:
        return read_json(self.assets_root / "config.default.json")

    def _ensure_runtime_ignored(self) -> None:
        ignore_path = self.project_root / ".gitignore"
        existing = ignore_path.read_text(encoding="utf-8") if ignore_path.exists() else ""
        lines = existing.splitlines()
        if ".orchestra/runtime/" not in lines:
            updated = existing
            if updated and not updated.endswith("\n"):
                updated += "\n"
            updated += ".orchestra/runtime/\n"
            atomic_write_text(ignore_path, updated)

    def bootstrap(self) -> dict[str, Any]:
        default = self._default_config()
        validate_config(default)
        created = False
        if self.config_path.exists():
            current = read_json(self.config_path)
            validate_config(current)
        else:
            atomic_write_json(self.config_path, default)
            created = True
        self._ensure_runtime_ignored()
        return {"created": created, "config": str(self.config_path)}

    def create_task(self, task: dict[str, Any]) -> dict[str, Any]:
        config = read_json(self.config_path)
        validate_task(task, config)
        if Path(task["git"]["worktree"]).resolve() != self.project_root:
            raise ConflictError("task worktree does not match project root")
        task_root = ensure_inside(self.control_root, self.control_root / "tasks" / task["task_id"])
        runtime_root = ensure_inside(self.control_root, self.control_root / "runtime" / task["task_id"])
        task_path = task_root / "task.json"
        state_path = runtime_root / "state.json"
        if task_path.exists():
            if read_json(task_path) != task:
                raise ConflictError("task_id already exists with different content")
            return read_json(state_path)
        atomic_write_json(task_path, task)
        state = new_state(task)
        atomic_write_json(state_path, state)
        return state
```

```python
from __future__ import annotations

from .constants import CORE_AGENTS
from .errors import ValidationError

SPECIALIST_ORDER = (
    "testing", "security", "migration", "ui", "dependency", "deployment"
)
RISK_TO_AGENT = {
    "behavior-change": "testing",
    "test-gap": "testing",
    "auth": "security",
    "authorization": "security",
    "privacy": "security",
    "secrets": "security",
    "untrusted-input": "security",
    "data-migration": "migration",
    "schema-change": "migration",
    "storage-change": "migration",
    "file-structure-change": "migration",
    "ui": "ui",
    "responsive": "ui",
    "accessibility": "ui",
    "visual-acceptance": "ui",
    "dependency-add": "dependency",
    "dependency-upgrade": "dependency",
    "ci": "deployment",
    "deployment": "deployment",
    "runtime-config": "deployment",
    "infrastructure": "deployment",
}


def route_agents(
    level: str,
    risk_tags: list[str],
    add: list[str],
    remove: list[str],
    reason: str = "",
    enabled_specialists: set[str] | None = None,
) -> dict[str, list[str]]:
    if level not in CORE_AGENTS:
        raise ValidationError(f"unknown task level: {level}")
    routed = set(CORE_AGENTS[level])
    reasons: list[str] = []
    for tag in risk_tags:
        if tag.startswith("specialist:") and len(tag.split(":", 1)[1]) > 0:
            agent = tag
        else:
            agent = RISK_TO_AGENT.get(tag)
        if agent is None:
            raise ValidationError(f"unknown risk tag: {tag}")
        routed.add(agent)
        reasons.append(f"{tag}->{agent}")
    routed.update(add)
    removed_core = set(CORE_AGENTS[level]) & set(remove)
    if removed_core and not reason.strip():
        raise ValidationError("removing a core Agent requires an owner-approved reason")
    routed.difference_update(remove)
    enabled = (
        {*SPECIALIST_ORDER, "specialist"}
        if enabled_specialists is None
        else enabled_specialists
    )
    for agent in routed:
        if agent in {"coding", "review", "codex_acceptance"}:
            continue
        capability = "specialist" if agent.startswith("specialist:") else agent
        if capability not in enabled:
            raise ValidationError(f"specialist is disabled by project config: {agent}")
    order = ["coding", "review", *SPECIALIST_ORDER]
    known = [name for name in order if name in routed]
    generic = sorted(name for name in routed if name.startswith("specialist:"))
    tail = ["codex_acceptance"] if "codex_acceptance" in routed else []
    return {"agents": known + generic + tail, "reasons": reasons}
```

```python
from __future__ import annotations

from pathlib import Path
from string import Template
from typing import Any

from .io import atomic_write_text, sha256_file


def _bullets(values: list[str], empty: str = "- None") -> str:
    return "\n".join(f"- {value}" for value in values) if values else empty


def _read_first(task: dict[str, Any]) -> list[str]:
    values = ["AGENTS.md", f".orchestra/tasks/{task['task_id']}/task.json"]
    values.extend(path for path in task["approved_basis"].values() if path)
    return list(dict.fromkeys(values))


def _write_template(template: Path, destination: Path, values: dict[str, str]) -> None:
    rendered = Template(template.read_text(encoding="utf-8")).substitute(values)
    atomic_write_text(destination, rendered.rstrip() + "\n")


def render_instructions(
    task: dict[str, Any],
    state: dict[str, Any],
    assets_root: Path,
    task_root: Path,
) -> dict[str, Any]:
    agent_root = task_root / "agents"
    common = {
        "task_id": task["task_id"],
        "goal": task["goal"],
        "read_first": _bullets(_read_first(task)),
        "allowed": _bullets(task["allowed_changes"]),
        "forbidden": _bullets(task["forbidden_changes"]),
        "relevant_files": _bullets(task["relevant_files"]),
        "acceptance": _bullets(task["acceptance_criteria"]),
        "checks": _bullets(task["required_checks"]),
        "base_commit": state["evidence"].get("base_commit", task["git"]["code_base_commit"]),
        "head_commit": state["evidence"].get("head_commit", "NOT_READY"),
    }
    generated: dict[str, str] = {}
    for agent in task["agents"]:
        if agent == "coding":
            name = "coding-fix-1.md" if state["iterations"]["coding"] else "coding.md"
            template = assets_root / "agents/coding.md.tmpl"
            values = {
                **common,
                "mode": "FIX" if state["iterations"]["coding"] else "IMPLEMENT",
                "result_path": f".orchestra/runtime/{task['task_id']}/results/coding.json",
            }
        elif agent == "review":
            name = "review.md"
            template = assets_root / "agents/review.md.tmpl"
            values = {
                **common,
                "result_path": f".orchestra/runtime/{task['task_id']}/results/review.json",
            }
        elif agent == "codex_acceptance":
            name = "codex-acceptance.md"
            template = assets_root / "agents/codex-acceptance.md.tmpl"
            values = {**common, "trigger": state["evidence"].get("codex_trigger", "owner-approved trigger")}
        else:
            specialty = agent.split(":", 1)[-1]
            name = f"{specialty}.md"
            template = assets_root / "agents/specialist.md.tmpl"
            values = {
                **common,
                "specialty": specialty,
                "questions": _bullets([f"Review the task's {specialty} risk against its acceptance criteria."]),
                "result_path": f".orchestra/runtime/{task['task_id']}/results/{specialty}.json",
            }
        destination = agent_root / name
        _write_template(template, destination, values)
        project_root = task_root.parents[2]
        generated[destination.relative_to(project_root).as_posix()] = sha256_file(destination)
    changed = dict(state)
    changed["generated_files"] = generated
    return changed
```

```python
from __future__ import annotations

import subprocess
from pathlib import Path, PurePosixPath
from typing import Any

from .errors import GitStateError, NotReadyError


def _git(root: Path, *args: str) -> str:
    try:
        result = subprocess.run(
            ["git", *args],
            cwd=root,
            check=True,
            text=True,
            capture_output=True,
        )
    except (OSError, subprocess.CalledProcessError) as exc:
        raise GitStateError(f"git command failed: {' '.join(args)}") from exc
    return result.stdout.strip()


def _path_allowed(path: str, rules: list[str]) -> bool:
    candidate = PurePosixPath(path)
    for rule in rules:
        if rule.endswith("/**"):
            root = PurePosixPath(rule[:-3])
            if candidate == root or root in candidate.parents:
                return True
        elif candidate == PurePosixPath(rule):
            return True
    return False


def validate_handoff(
    project_root: Path,
    task: dict[str, Any],
    base_commit: str,
    head_commit: str,
) -> dict[str, Any]:
    actual_root = Path(_git(project_root, "rev-parse", "--show-toplevel")).resolve()
    if actual_root != project_root.resolve():
        raise GitStateError("worktree root mismatch")
    if actual_root != Path(task["git"]["worktree"]).resolve():
        raise GitStateError("task worktree mismatch")
    branch = _git(project_root, "branch", "--show-current")
    if branch != task["git"]["task_branch"]:
        raise GitStateError(f"branch mismatch: {branch}")
    if _git(project_root, "status", "--porcelain"):
        raise NotReadyError("task worktree is not clean")
    try:
        _git(project_root, "merge-base", "--is-ancestor", base_commit, head_commit)
    except GitStateError as exc:
        raise GitStateError("base commit is not an ancestor of head commit") from exc
    if _git(project_root, "rev-parse", "HEAD") != head_commit:
        raise GitStateError("head commit does not match worktree HEAD")
    changed = [
        line for line in _git(project_root, "diff", "--name-only", f"{base_commit}...{head_commit}").splitlines()
        if line
    ]
    for path in changed:
        if _path_allowed(path, task["forbidden_changes"]):
            raise GitStateError(f"forbidden path changed: {path}")
        if not _path_allowed(path, task["allowed_changes"]):
            raise GitStateError(f"out-of-scope path changed: {path}")
    return {
        "base_commit": base_commit,
        "head_commit": head_commit,
        "changed_files": changed,
        "branch": branch,
        "clean": True,
    }
```

```python
from __future__ import annotations

import argparse

def parser() -> argparse.ArgumentParser:
    root = argparse.ArgumentParser(prog="orchestra")

    commands = root.add_subparsers(dest="command", required=True)

    ingest.add_argument("--task-id", required=True)

    revise = commands.add_parser("revise")
    revise.add_argument("--task-id", required=True)
    revise.add_argument("--task-file", required=True, type=Path)

def main(argv: list[str] | None = None) -> int:

    service = OrchestraService(args.project, skill_root / "assets/orchestration")
    try:
        if args.command == "bootstrap":
            result = service.bootstrap()
        elif args.command == "create-task":
            result = service.create_task(read_json(args.task_file))
        elif args.command == "status":

        elif args.command == "render":

        elif args.command == "ingest":
            result = service.ingest(args.task_id, args.role, args.result_file)
        elif args.command == "revise":
```
